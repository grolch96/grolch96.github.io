💻 Скрипты системного администрирования (Linux + Windows)

Сборник скриптов для автоматизации рутинных задач системного администратора. Все примеры проверены, содержат обработку ошибок и режим безопасного тестирования (dry-run), чтобы не сломать систему при первом запуске.

📌 Принципы: безопасность (никаких изменений без подтверждения), идемпотентность (повторный запуск не ломает систему), логирование (всё пишется в лог), обработка ошибок (скрипт не падает, а сообщает о проблеме).

1. Резервное копирование файлов и базы данных

Автоматическое резервное копирование важных данных с ротацией старых архивов (храним последние N дней).

🐧 Linux (Bash)

#!/bin/bash
# backup.sh - резервное копирование с ротацией
BACKUP_DIR="/backup"
SOURCE_DIR="/var/www/octopus"
DATE=$(date +%Y-%m-%d)
KEEP_DAYS=7

# Проверка прав
if [[ $EUID -ne 0 ]]; then
    echo "Ошибка: запустите от root"
    exit 1
fi

# Создание архива
tar -czf "$BACKUP_DIR/backup_$DATE.tar.gz" "$SOURCE_DIR" 2>/dev/null || {
    echo "Ошибка: не удалось создать архив"
    exit 1
}

# Удаление старых архивов
find "$BACKUP_DIR" -name "backup_*.tar.gz" -mtime +$KEEP_DAYS -delete

echo "✅ Резервное копирование завершено: $BACKUP_DIR/backup_$DATE.tar.gz"

🪟 Windows (CMD/Batch)

:: backup.cmd - резервное копирование с ротацией
@echo off
set BACKUP_DIR=D:\Backup
set SOURCE_DIR=C:\inetpub\wwwroot
set DATE=%date:~-4%-%date:~3,2%-%date:~0,2%

:: Проверка прав администратора
net session >nul 2>&1 || (echo Ошибка: запустите от администратора & exit /b 1)

:: Создание архива через 7-Zip
"C:\Program Files\7-Zip\7z.exe" a -tzip "%BACKUP_DIR%\backup_%DATE%.zip" "%SOURCE_DIR%" >nul
if %errorlevel% NEQ 0 (echo Ошибка: не удалось создать архив & exit /b 1)

:: Удаление старых архивов (старше 7 дней)
forfiles /p "%BACKUP_DIR%" /m backup_*.zip /d -7 /c "cmd /c del @path" 2>nul

echo ✅ Резервное копирование завершено: %BACKUP_DIR%\backup_%DATE%.zip

2. Мониторинг дискового пространства и нагрузки

Проверка свободного места на дисках и отправка уведомления при превышении порога (например, 80%).

🐧 Linux (Bash)

#!/bin/bash
# monitor_disk.sh - проверка свободного места
THRESHOLD=80
ALERT_EMAIL="admin@example.com"

# Проверка всех смонтированных разделов
df -H | grep -vE '^Filesystem|tmpfs|cdrom' | awk '{ print $5 " " $1 }' | while read output; do
    usage=$(echo $output | awk '{ print $1}' | cut -d'%' -f1)
    partition=$(echo $output | awk '{ print $2 }')
    
    if [[ $usage -ge $THRESHOLD ]]; then
        echo "⚠️ ВНИМАНИЕ: раздел $partition заполнен на $usage%"
        echo "Раздел $partition заполнен на $usage%" | mail -s "Disk Alert: $partition" "$ALERT_EMAIL"
    fi
done

🪟 Windows (CMD/Batch)

:: monitor_disk.cmd - проверка свободного места
@echo off
setlocal enabledelayedexpansion
set THRESHOLD=80

:: Получение информации о дисках
for /f "skip=1 tokens=1,2,3" %%a in ('wmic logicaldisk get caption^,size^,freespace') do (
    set drive=%%a
    set size=%%b
    set free=%%c
    
    if not "!size!"=="0" (
        set /a usage=100 - (!free! * 100 / !size!)
        if !usage! GEQ %THRESHOLD% (
            echo ⚠️ ВНИМАНИЕ: диск !drive! заполнен на !usage!%%
        )
    )
)

3. Массовое управление пользователями и группами

Создание пользователей, добавление в группы, блокировка неактивных учётных записей.

🐧 Linux (Bash)

#!/bin/bash
# user_manage.sh - массовое управление пользователями
USER_LIST="/etc/users.list"  # формат: username:group

# Проверка прав
[[ $EUID -ne 0 ]] && { echo "Ошибка: запустите от root"; exit 1; }

# Чтение списка и создание пользователей
while IFS=':' read -r username group; do
    if ! id "$username" &>/dev/null; then
        useradd -m -s /bin/bash "$username"
        echo "Создан пользователь: $username"
    else
        echo "Пользователь уже существует: $username"
    fi
    
    # Добавление в группу
    if [[ -n "$group" ]]; then
        usermod -aG "$group" "$username"
        echo "Пользователь $username добавлен в группу $group"
    fi
done < "$USER_LIST"

🪟 Windows (CMD/Batch)

:: user_manage.cmd - массовое создание пользователей
@echo off
set USER_LIST=C:\users.list

:: Проверка прав администратора
net session >nul 2>&1 || (echo Ошибка: запустите от администратора & exit /b 1)

:: Чтение списка и создание пользователей
for /f "tokens=1,2 delims=:" %%a in (%USER_LIST%) do (
    net user %%a /add >nul 2>&1
    if !errorlevel! EQU 0 (echo Создан пользователь: %%a) else (echo Пользователь уже существует: %%a)
    
    if not "%%b"=="" (
        net localgroup %%b %%a /add >nul 2>&1
        echo Пользователь %%a добавлен в группу %%b
    )
)

4. Режим безопасного тестирования (dry-run)

Любой скрипт можно запустить в тестовом режиме, чтобы проверить его логику, не внося изменений в систему.

🐧 Linux (Bash)

#!/bin/bash
# safe_script.sh - с поддержкой dry-run
TEST_MODE="${TEST_MODE:-false}"

log_action() {
    if [[ "$TEST_MODE" == "true" ]]; then
        echo "🔬 [ТЕСТ] $1"
    else
        echo "✅ [ВЫПОЛНЕНО] $1"
        eval "$2"  # реальное выполнение
    fi
}

# Пример использования
log_action "Создание резервной копии" "tar -czf /backup/data.tar.gz /var/www"
log_action "Перезапуск сервиса" "systemctl restart nginx"

🪟 Windows (CMD/Batch)

:: safe_script.cmd - с поддержкой dry-run
@echo off
set TEST_MODE=%TEST_MODE%
if "%TEST_MODE%"=="" set TEST_MODE=false

:: Пример вызова с параметром
call :log_action "Создание архива" "7z a backup.zip C:\data"
call :log_action "Перезапуск сервиса" "net stop /y spooler & net start spooler"
goto :eof

:log_action
if /i "%TEST_MODE%"=="true" (
    echo 🔬 [ТЕСТ] %~1
) else (
    echo ✅ [ВЫПОЛНЕНО] %~1
    %~2
)
goto :eof

5. Управление службами (запуск, остановка, перезапуск)

Скрипты для проверки статуса служб и автоматического перезапуска при падении. Работают и в Linux (systemctl), и в Windows (sc/net).

🐧 Linux (Bash)

#!/bin/bash
# service_check.sh - проверка и перезапуск службы
SERVICE="nginx"
MAX_ATTEMPTS=3
ATTEMPT=1

while [[ $ATTEMPT -le $MAX_ATTEMPTS ]]; do
    if systemctl is-active --quiet "$SERVICE"; then
        echo "✅ Служба $SERVICE работает"
        exit 0
    else
        echo "⚠️ Служба $SERVICE не работает (попытка $ATTEMPT/$MAX_ATTEMPTS)"
        systemctl restart "$SERVICE" 2>/dev/null
        sleep 3
        ((ATTEMPT++))
    fi
done

echo "❌ Не удалось запустить службу $SERVICE после $MAX_ATTEMPTS попыток"
echo "Проверьте логи: journalctl -u $SERVICE -n 20"
exit 1

🪟 Windows (CMD/Batch)

:: service_check.cmd - проверка и перезапуск службы
@echo off
set SERVICE=Spooler
set MAX_ATTEMPTS=3
set ATTEMPT=1

:check_loop
sc query "%SERVICE%" | find "RUNNING" >nul
if %errorlevel% EQU 0 (
    echo ✅ Служба %SERVICE% работает
    goto :eof
) else (
    echo ⚠️ Служба %SERVICE% не работает (попытка %ATTEMPT%/%MAX_ATTEMPTS%)
    net start "%SERVICE%" >nul 2>&1
    timeout /t 3 /nobreak >nul
    set /a ATTEMPT+=1
    if %ATTEMPT% LEQ %MAX_ATTEMPTS% goto check_loop
)

echo ❌ Не удалось запустить службу %SERVICE% после %MAX_ATTEMPTS% попыток
echo Проверьте логи: eventvwr.msc

6. Работа с логами (анализ и ротация)

Поиск ошибок в логах за последние сутки и автоматическая ротация (архивирование старых логов).

🐧 Linux (Bash)

#!/bin/bash
# log_analyze.sh - анализ и ротация логов
LOG_DIR="/var/log"
ARCHIVE_DIR="/var/log/archive"
DATE=$(date +%Y-%m-%d)

# 1. Поиск ошибок за последние 24 часа
echo "=== Ошибки за последние 24 часа ==="
grep -i "error\|fail\|critical" "$LOG_DIR/syslog" 2>/dev/null | tail -20

# 2. Ротация: архивируем логи старше 7 дней
mkdir -p "$ARCHIVE_DIR"
find "$LOG_DIR" -name "*.log" -mtime +7 -exec gzip {} \; -exec mv {}.gz "$ARCHIVE_DIR/" \;

# 3. Удаление архивов старше 30 дней
find "$ARCHIVE_DIR" -name "*.gz" -mtime +30 -delete

echo "✅ Анализ и ротация логов завершены"

🪟 Windows (CMD/Batch)

:: log_analyze.cmd - анализ и ротация логов
@echo off
set LOG_DIR=C:\Logs
set ARCHIVE_DIR=C:\Logs\archive
set DATE=%date:~-4%-%date:~3,2%-%date:~0,2%

:: 1. Поиск ошибок в логах
echo === Ошибки в логах ===
findstr /i /c:"error" /c:"fail" "%LOG_DIR%\*.log" 2>nul | more

:: 2. Ротация: архивируем логи старше 7 дней
if not exist "%ARCHIVE_DIR%" mkdir "%ARCHIVE_DIR%"
forfiles /p "%LOG_DIR%" /m *.log /d -7 /c "cmd /c move @path %ARCHIVE_DIR%\" 2>nul

:: 3. Удаление архивов старше 30 дней
forfiles /p "%ARCHIVE_DIR%" /m *.log /d -30 /c "cmd /c del @path" 2>nul

echo ✅ Анализ и ротация логов завершены

7. Сетевые проверки (пинг, порты, трассировка)

Автоматическая проверка доступности узлов, открытых портов и пути следования пакетов.

🐧 Linux (Bash)

#!/bin/bash
# network_check.sh - проверка сети
HOSTS=("192.168.1.1" "8.8.8.8" "google.com")
PORTS=(80 443 22)

# 1. Пинг узлов
echo "=== Проверка доступности ==="
for host in "${HOSTS[@]}"; do
    if ping -c 1 -W 2 "$host" &>/dev/null; then
        echo "✅ $host доступен"
    else
        echo "❌ $host недоступен"
    fi
done

# 2. Проверка портов
echo "=== Проверка портов ==="
for port in "${PORTS[@]}"; do
    if timeout 1 bash -c "cat < /dev/null > /dev/tcp/8.8.8.8/$port" 2>/dev/null; then
        echo "✅ Порт $port открыт"
    else
        echo "❌ Порт $port закрыт"
    fi
done

# 3. Трассировка
echo "=== Трассировка до 8.8.8.8 ==="
traceroute -m 10 -w 1 8.8.8.8 2>/dev/null | head -10

🪟 Windows (CMD/Batch)

:: network_check.cmd - проверка сети
@echo off
set HOSTS=192.168.1.1 8.8.8.8 google.com

echo === Проверка доступности ===
for %%h in (%HOSTS%) do (
    ping -n 1 -w 2000 %%h >nul 2>&1
    if !errorlevel! EQU 0 (echo ✅ %%h доступен) else (echo ❌ %%h недоступен)
)

echo === Проверка портов ===
for %%p in (80 443 22) do (
    powershell -Command "Test-NetConnection -ComputerName 8.8.8.8 -Port %%p -InformationLevel Quiet" >nul 2>&1
    if !errorlevel! EQU 0 (echo ✅ Порт %%p открыт) else (echo ❌ Порт %%p закрыт)
)

echo === Трассировка до 8.8.8.8 ===
tracert -h 10 -w 1000 8.8.8.8

8. Инвентаризация железа (сбор информации о системе)

Сбор данных о процессоре, памяти, дисках, сетевых интерфейсах и сохранение в отчёт для дальнейшего анализа.

🐧 Linux (Bash)

#!/bin/bash
# inventory.sh - сбор информации о системе
REPORT="/tmp/inventory_$(hostname)_$(date +%Y%m%d).txt"

{
    echo "=== ИНВЕНТАРИЗАЦИЯ СИСТЕМЫ ==="
    echo "Дата: $(date)"
    echo "Хост: $(hostname)"
    echo "ОС: $(lsb_release -d 2>/dev/null | cut -f2 || uname -s)"
    echo ""
    
    echo "=== CPU ==="
    lscpu | grep -E "Model name|CPU\(s\)|MHz" 2>/dev/null || echo "lscpu не доступен"
    echo ""
    
    echo "=== Память ==="
    free -h 2>/dev/null || echo "free не доступен"
    echo ""
    
    echo "=== Диски ==="
    lsblk -o NAME,SIZE,TYPE,MOUNTPOINT 2>/dev/null || echo "lsblk не доступен"
    echo ""
    
    echo "=== Сетевые интерфейсы ==="
    ip -br a 2>/dev/null || echo "ip не доступен"
    echo ""
    
    echo "=== Открытые порты ==="
    ss -tuln 2>/dev/null | head -20 || echo "ss не доступен"
} > "$REPORT"

echo "✅ Отчёт сохранён: $REPORT"
cat "$REPORT"

9. Обработка ошибок и логирование

Правильная обработка ошибок: скрипт не «падает» молча, а пишет понятное сообщение в лог и продолжает работу (или корректно завершается).

🐧 Linux (Bash)

#!/bin/bash
# error_handling.sh - обработка ошибок и логирование
LOG_FILE="/var/log/my_script.log"
MAX_RETRIES=3

log() {
    local level="$1"
    local message="$2"
    echo "[$(date '+%Y-%m-%d %H:%M:%S')] [$level] $message" | tee -a "$LOG_FILE"
}

# Функция с повторными попытками
retry_command() {
    local cmd="$1"
    local attempt=1
    
    while [[ $attempt -le $MAX_RETRIES ]]; do
        if eval "$cmd" 2>/dev/null; then
            log "INFO" "Команда выполнена: $cmd"
            return 0
        else
            log "WARN" "Попытка $attempt/$MAX_RETRIES не удалась: $cmd"
            sleep 2
            ((attempt++))
        fi
    done
    
    log "ERROR" "Команда не выполнена после $MAX_RETRIES попыток: $cmd"
    return 1
}

# Пример использования
if retry_command "systemctl restart nginx"; then
    log "SUCCESS" "Nginx перезапущен"
else
    log "ERROR" "Не удалось перезапустить Nginx"
    exit 1
fi

🪟 Windows (CMD/Batch)

:: error_handling.cmd - обработка ошибок и логирование
@echo off
setlocal enabledelayedexpansion
set LOG_FILE=C:\Logs\my_script.log
set MAX_RETRIES=3

:: Функция логирования
call :log "INFO" "Скрипт запущен"

:: Функция с повторными попытками
call :retry "net start Spooler"
if !errorlevel! EQU 0 (
    call :log "SUCCESS" "Служба Spooler запущена"
) else (
    call :log "ERROR" "Не удалось запустить Spooler"
    exit /b 1
)
goto :eof

:log
echo [%date% %time%] [%~1] %~2 >> "%LOG_FILE%"
echo [%date% %time%] [%~1] %~2
goto :eof

:retry
set CMD=%~1
set ATTEMPT=1
:retry_loop
%CMD% >nul 2>&1
if !errorlevel! EQU 0 (
    call :log "INFO" "Команда выполнена: %CMD%"
    exit /b 0
) else (
    call :log "WARN" "Попытка !ATTEMPT!/%MAX_RETRIES% не удалась: %CMD%"
    timeout /t 2 /nobreak >nul
    set /a ATTEMPT+=1
    if !ATTEMPT! LEQ %MAX_RETRIES% goto retry_loop
)
call :log "ERROR" "Команда не выполнена после %MAX_RETRIES% попыток: %CMD%"
exit /b 1

10. Автоматизация через планировщик (cron / Task Scheduler)

Настройка автоматического запуска скриптов по расписанию: ежедневно, еженедельно или по событию.

🐧 Linux (Bash / cron)

#!/bin/bash
# schedule_task.sh - добавление задачи в cron
SCRIPT_PATH="/usr/local/bin/backup.sh"
CRON_SCHEDULE="0 2 * * *"  # каждый день в 2:00
LOG_FILE="/var/log/cron_backup.log"

# Проверка прав
[[ $EUID -ne 0 ]] && { echo "Ошибка: запустите от root"; exit 1; }

# Проверка существования скрипта
if [[ ! -f "$SCRIPT_PATH" ]]; then
    echo "Ошибка: скрипт $SCRIPT_PATH не найден"
    exit 1
fi

# Добавление задачи в cron (если ещё нет)
if crontab -l 2>/dev/null | grep -q "$SCRIPT_PATH"; then
    echo "⚠️ Задача уже существует в cron"
else
    (crontab -l 2>/dev/null; echo "$CRON_SCHEDULE $SCRIPT_PATH >> $LOG_FILE 2>&1") | crontab -
    echo "✅ Задача добавлена в cron: $CRON_SCHEDULE $SCRIPT_PATH"
fi

# Показать текущее расписание
echo "=== Текущее расписание cron ==="
crontab -l

🪟 Windows (CMD/Batch / Task Scheduler)

:: schedule_task.cmd - добавление задачи в Планировщик
@echo off
set SCRIPT_PATH=C:\Scripts\backup.cmd
set TASK_NAME=DailyBackup

:: Проверка прав администратора
net session >nul 2>&1 || (echo Ошибка: запустите от администратора & exit /b 1)

:: Проверка существования скрипта
if not exist "%SCRIPT_PATH%" (
    echo Ошибка: скрипт %SCRIPT_PATH% не найден
    exit /b 1
)

:: Создание задачи (ежедневно в 2:00)
schtasks /create /tn "%TASK_NAME%" /tr "%SCRIPT_PATH%" /sc daily /st 02:00 /ru SYSTEM /f
if %errorlevel% EQU 0 (
    echo ✅ Задача добавлена: %TASK_NAME% (ежедневно в 2:00)
) else (
    echo ❌ Ошибка при создании задачи
    exit /b 1
)

:: Показать текущее расписание
echo === Текущие задачи ===
schtasks /query /tn "%TASK_NAME%" /v /fo list

11. Проверка безопасности (базовый аудит)

Быстрая проверка системы на наличие типичных проблем: открытые порты, слабые пароли, устаревшие пакеты, неактивные учётные записи.

🐧 Linux (Bash)

#!/bin/bash
# security_audit.sh - базовый аудит безопасности
LOG_FILE="/var/log/security_audit.log"

log() { echo "[$(date '+%Y-%m-%d %H:%M:%S')] $1" | tee -a "$LOG_FILE"; }

log "=== АУДИТ БЕЗОПАСНОСТИ ==="

# 1. Пользователи без пароля
log "1. Пользователи без пароля:"
awk -F: '($2 == "" || $2 == "!") {print $1}' /etc/shadow 2>/dev/null || log "  Нет доступа к /etc/shadow"

# 2. Пользователи с UID 0 (кроме root)
log "2. Пользователи с UID 0:"
awk -F: '($3 == 0) {print $1}' /etc/passwd

# 3. Открытые порты
log "3. Открытые порты (наружу):"
ss -tuln 2>/dev/null | grep -E "0.0.0.0|:::" | head -10

# 4. Устаревшие пакеты
log "4. Пакеты, требующие обновления:"
apt list --upgradable 2>/dev/null | tail -n +2 | head -10 || log "  apt не доступен"

# 5. Последние входы в систему
log "5. Последние входы (топ-5):"
last -n 5 2>/dev/null | head -5

log "=== АУДИТ ЗАВЕРШЁН ==="

🪟 Windows (CMD/Batch)

:: security_audit.cmd - базовый аудит безопасности
@echo off
set LOG_FILE=C:\Logs\security_audit.log

echo [%date% %time%] === АУДИТ БЕЗОПАСНОСТИ === >> "%LOG_FILE%"

echo 1. Пользователи-администраторы:

14. Управление питанием (перезагрузка, выключение, гибернация)

Скрипты для автоматического выключения, перезагрузки и перевода компьютеров в спящий режим по расписанию или команде.

🐧 Linux (Bash)

#!/bin/bash
# power_manage.sh - управление питанием
ACTION="${1:-status}"

case "$ACTION" in
    reboot)
        echo "🔄 Перезагрузка системы через 5 секунд..."
        shutdown -r +0
        ;;
    shutdown)
        echo "⏻ Выключение системы через 5 секунд..."
        shutdown -h +0
        ;;
    suspend)
        echo "💤 Перевод в спящий режим..."
        systemctl suspend
        ;;
    hibernate)
        echo "💤 Гибернация..."
        systemctl hibernate
        ;;
    status)
        echo "=== Статус питания ==="
        uptime -p 2>/dev/null || uptime
        echo ""
        echo "Текущее время: $(date)"
        echo "Последняя загрузка: $(who -b 2>/dev/null | awk '{print $3, $4}')"
        ;;
    *)
        echo "Использование: $0 {reboot|shutdown|suspend|hibernate|status}"
        exit 1
        ;;
esac

🪟 Windows (CMD/Batch)

15. Сетевые диски (подключение, отключение, проверка)

Скрипты для автоматического подключения сетевых дисков (SMB/CIFS) с проверкой доступности и переподключением при сбое.

🐧 Linux (Bash)

#!/bin/bash
# mount_network_drive.sh - подключение сетевого диска
MOUNT_POINT="/mnt/share"
REMOTE_SHARE="//server.example.com/share"
USERNAME="domain\\user"
PASSWORD="password"
MAX_RETRIES=3
ATTEMPT=1

# Проверка, смонтирован ли уже
if mountpoint -q "$MOUNT_POINT"; then
    echo "✅ Диск уже смонтирован: $MOUNT_POINT"
    df -h "$MOUNT_POINT" | tail -1
    exit 0
fi

# Создание точки монтирования
sudo mkdir -p "$MOUNT_POINT"

# Попытки монтирования
while [[ $ATTEMPT -le $MAX_RETRIES ]]; do
    if sudo mount -t cifs "$REMOTE_SHARE" "$MOUNT_POINT" \
        -o username="$USERNAME",password="$PASSWORD",vers=3.0,uid=$(id -u),gid=$(id -g); then
        echo "✅ Диск успешно смонтирован: $MOUNT_POINT"
        df -h "$MOUNT_POINT" | tail -1
        exit 0
    else
        echo "⚠️ Попытка $ATTEMPT/$MAX_RETRIES не удалась"
        sleep 2
        ((ATTEMPT++))
    fi
done

echo "❌ Не удалось смонтировать диск после $MAX_RETRIES попыток"
echo "Проверьте: ping server.example.com, доступность порта 445"
exit 1

📌 Требуется пакет cifs-utils: sudo apt install cifs-utils

🪟 Windows (CMD/Batch)

:: mount_network_drive.cmd - подключение сетевого диска
@echo off

set REMOTE_SHARE=\\server\share
set USERNAME=domain\user

    goto :eof
)

:: Отключение старого (если был)

:: Подключение

    dir %DRIVE% | find "bytes free"
) else (
    echo ❌ Не удалось подключить диск %DRIVE%
    echo Проверьте: ping server, доступность порта 445
    exit /b 1
)

16. SMTP-уведомления (отправка почты при сбоях)

Автоматическая отправка писем администратору при срабатывании тревоги (ошибка в логах, переполнение диска, падение службы).

🐧 Linux (Bash / msmtp или mail)

#!/bin/bash
# send_alert.sh - отправка уведомления
ALERT_EMAIL="admin@example.com"
SUBJECT="ALERT: System Issue on $(hostname)"
BODY="Обнаружена проблема: $1"
LOG_FILE="/var/log/alerts.log"

# Логирование
echo "[$(date '+%Y-%m-%d %H:%M:%S')] $BODY" >> "$LOG_FILE"

# Отправка через msmtp (лёгкий SMTP-клиент)
if command -v msmtp &>/dev/null; then
    echo -e "Subject: $SUBJECT\n\n$BODY" | msmtp "$ALERT_EMAIL"
    echo "✅ Письмо отправлено через msmtp"
# Или через mail (если установлен mailutils)
elif command -v mail &>/dev/null; then
    echo "$BODY" | mail -s "$SUBJECT" "$ALERT_EMAIL"
    echo "✅ Письмо отправлено через mail"
else
    echo "❌ Не найден SMTP-клиент (msmtp или mail)"
    echo "Установите: sudo apt install msmtp msmtp-mta"
    exit 1
fi

📌 Установка: sudo apt install msmtp msmtp-mta
📌 Настройка: ~/.msmtprc (SMTP-сервер, логин, пароль)

🪟 Windows (CMD/Batch / blat)

    if %errorlevel% EQU 0 (
        echo ✅ Письмо отправлено через blat
    ) else (
        echo ❌ Ошибка отправки через blat
        exit /b 1
    )
) else (
    echo ❌ Не найден blat.exe
    echo Скачайте: https://www.blat.net/
    exit /b 1
)
